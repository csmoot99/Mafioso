set -eu
mkdir -p "$RUNNER_TEMP/fm-test"
# Most lanes upload their timing JSON at the artifact root; the Herdr
# artifact also carries diagnostics from a second directory, so its
# timing JSON sits under fm-test/. Find them wherever they landed.
inputs=()
while IFS= read -r input; do
  inputs+=("$input")
done < <(find "$RUNNER_TEMP/fm-test-aggregate" -type f -name 'fm-test-timing-*.json' | LC_ALL=C sort)
if [ "${#inputs[@]}" -eq 0 ]; then
  echo "::warning::no per-lane timing JSON found to aggregate"
  exit 0
fi
bin/fm-test-run.sh --aggregate-json \
  "$RUNNER_TEMP/fm-test/fm-test-timing-aggregate.json" \
  "${inputs[@]}"
